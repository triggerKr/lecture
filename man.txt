7. 실제 Spring Boot Controller
@PostMapping("/login")
public ResponseEntity<LoginResponse> login(
        HttpServletResponse response) {

    String accessToken =
            jwtProvider.createAccessToken(user);

    String refreshToken =
            jwtProvider.createRefreshToken(user);

    redisService.save(user.getId(), refreshToken);

    ResponseCookie cookie =
            ResponseCookie.from("refreshToken", refreshToken)
                    .httpOnly(true)
                    .secure(true)
                    .sameSite("Strict")
                    .path("/")
                    .maxAge(Duration.ofDays(14))
                    .build();

    response.addHeader(
            HttpHeaders.SET_COOKIE,
            cookie.toString());

    LoginResponse dto =
            new LoginResponse(
                    accessToken,
                    user.getId(),
                    user.getName(),
                    user.getRole());

    return ResponseEntity.ok(dto);
}
